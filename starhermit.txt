# StarHermit engine script game: one package, two runtimes. Browsers open
# launch=index.html, the web loader `npm run package` writes (it pins the
# engine on wiki.starhermit.com by hash; never edit it); the StarHermit client
# plays engine.entry natively. Re-run `npm run package` after every change and
# commit index.html + starhermit-files.json (spec.md §13).
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.28.0
engine.minor=30
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
control.left=ArrowLeft | Move selection left
control.right=ArrowRight | Move selection right
control.up=ArrowUp | Move selection up
control.down=ArrowDown | Move selection down
control.build=Enter+NumpadEnter | Build with the current tool
control.tick=Space | Run one tick
control.auto=KeyA | Toggle auto-run
control.belt=KeyB | Belt tool
control.machine=KeyM | Machine tool
control.rotate=KeyR | Rotate tool
control.remove=KeyX | Remove tool
control.upgrade=KeyU | Upgrade tool
control.hint=KeyH | Hint
control.undo=KeyZ | Undo
control.pause=KeyP+Escape | Pause
engine.group.left=Board
engine.group.tick=Simulation
engine.group.belt=Tools
engine.group.hint=Assists
